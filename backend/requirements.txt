# Los rangos son estrechos a propósito (`>=X.Y,<X.Y+1`): fijan la serie, no la versión exacta, así
# que un parche de seguridad entra solo y un salto de serie exige una revisión explícita de las
# notas de versión. El costo es que Dependabot respeta el rango y nunca propone el salto — ampliarlo
# es una decisión manual. Ver `docs/seguridad_dependencias.md`.

# 5.2 es LTS (soporte extendido hasta abril de 2028). Se eligió por sobre 6.x justamente por eso:
# la 5.0 quedó sin parches de seguridad y el objetivo era volver a una serie soportada con el menor
# salto posible.
Django>=5.2,<5.3
# DRF 3.18 exige Django>=5.2, así que las dos versiones quedan acopladas a propósito.
djangorestframework>=3.18,<3.19
djangorestframework-simplejwt>=5.5,<5.6
django-cors-headers>=4.9,<4.10
# mssql-django <1.6 no soporta Django 5.1+; 1.8 es la primera serie que declara soporte de 5.2.
mssql-django>=1.8,<1.9
pyodbc>=5.1,<5.2
pandas>=2.2,<2.3
openpyxl>=3.1.5,<3.2
xlrd>=2.0,<2.1
python-dateutil>=2.9,<2.10
python-dotenv>=1.0,<1.1
argon2-cffi>=23.1,<26
# Pillow es la unica biblioteca del backend que parsea un binario controlado por el usuario (los
# avatares, via el ImageField de Django). La 10.4 acumulaba 12 avisos de seguridad, 9 de severidad
# alta, varios alcanzables desde `Image.open()`. Por eso su rango se mueve apenas sale un parche.
Pillow>=12.3,<12.4
requests>=2.32,<2.33
# Saneo del HTML de las plantillas de correo (`apps/authentication/sanitizacion.py`). Se eligió
# sobre `bleach` por dos motivos comprobados contra la plantilla real: `bleach` borra todos los
# `style` en línea salvo que se le configure un saneador de CSS aparte (y un correo sin estilos en
# línea no tiene diseño), y su propio proyecto lo declara en modo mantenimiento recomendando nh3.
nh3>=0.3,<0.4
